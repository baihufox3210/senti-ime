cmake_minimum_required(VERSION 3.21)
project(fcitx5-senti VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
find_package(Fcitx5Core 5.1.9 REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(JSONC REQUIRED IMPORTED_TARGET json-c>=0.15)
set(SENTI_HOST_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../../../target/release/senti-host" CACHE FILEPATH "Prebuilt Rust senti-host binary")
get_filename_component(SENTI_HOST_EXECUTABLE "${SENTI_HOST_EXECUTABLE}" ABSOLUTE)
if(NOT EXISTS "${SENTI_HOST_EXECUTABLE}")
    message(FATAL_ERROR "Build cargo build -p senti-platform --release first, or set SENTI_HOST_EXECUTABLE")
endif()
add_library(senti SHARED senti.cpp bridge.cpp)
set_target_properties(senti PROPERTIES PREFIX "")
target_compile_features(senti PUBLIC cxx_std_20)
target_compile_options(senti PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(senti PRIVATE Fcitx5::Core PkgConfig::JSONC Threads::Threads)
target_compile_definitions(senti PRIVATE SENTI_HOST_PATH="${CMAKE_INSTALL_FULL_LIBEXECDIR}/senti-ime/senti-host")
configure_file(senti-addon.conf.in senti-addon.conf @ONLY)
install(TARGETS senti LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/fcitx5")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/senti-addon.conf" DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/addon" RENAME senti.conf)
foreach(mode IN ITEMS mixed mixed-explicit zhuyin pinyin english)
    if(mode STREQUAL "mixed")
        set(IM_ID senti)
        set(IM_NAME "SentiIME — Mixed")
        set(IM_LABEL "中英")
        set(IM_NATIVE_NAME "SentiIME 中英混打")
    elseif(mode STREQUAL "mixed-explicit")
        set(IM_ID senti-mixed)
        set(IM_NAME "SentiIME — Mixed (Chinese on)")
        set(IM_LABEL "中英")
        set(IM_NATIVE_NAME "SentiIME 中英混打（中文起始）")
    else()
        set(IM_ID "senti-${mode}")
        set(IM_NAME "SentiIME — ${mode}")
        if(mode STREQUAL "zhuyin")
            set(IM_LABEL "注")
            set(IM_NATIVE_NAME "SentiIME 純注音")
        elseif(mode STREQUAL "pinyin")
            set(IM_LABEL "拼")
            set(IM_NATIVE_NAME "SentiIME 拼音")
        else()
            set(IM_LABEL "EN")
            set(IM_NATIVE_NAME "SentiIME 純英文")
        endif()
    endif()
    configure_file(senti-inputmethod.conf.in "${IM_ID}.conf" @ONLY)
    install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${IM_ID}.conf" DESTINATION "${CMAKE_INSTALL_DATADIR}/fcitx5/inputmethod")
endforeach()
install(FILES ../../../assets/branding/senti-ime.svg ../../../assets/branding/senti-ime-symbolic.svg
    DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps")
foreach(size IN ITEMS 16 22 24 32 48 64 128 256)
    install(FILES "../../../assets/branding/png/${size}x${size}/senti-ime.png"
        DESTINATION "${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps")
endforeach()
install(FILES "../../../assets/branding/png/48x48/senti-ime.png"
    DESTINATION "${CMAKE_INSTALL_DATADIR}/pixmaps" RENAME senti-ime-panel-v2.png)
install(PROGRAMS "${SENTI_HOST_EXECUTABLE}" DESTINATION "${CMAKE_INSTALL_LIBEXECDIR}/senti-ime")
install(FILES README.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/senti-ime" RENAME README-linux.md)
install(FILES ../../../docs/LOCAL_LANGUAGE_MODEL.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/senti-ime")
install(FILES ../../../crates/senti-predictors/data/SOURCES.md DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/senti-ime")
install(DIRECTORY ../../../crates/senti-predictors/data/licenses/ DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/senti-ime")
install(DIRECTORY ../../../models/tiny-ime-v2/licenses/ DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/senti-ime/neural-v2")
install(FILES ../../../models/tiny-ime-v2/README.md ../../../models/tiny-ime-v2/training-report.json
    ../../../models/tiny-ime-v2/full-vocabulary-evaluation.json DESTINATION "${CMAKE_INSTALL_DATADIR}/doc/senti-ime/neural-v2")
if(BUILD_TESTING)
    add_executable(senti-bad-host tests/bad-host.cpp)
    target_compile_features(senti-bad-host PRIVATE cxx_std_20)
    target_link_libraries(senti-bad-host PRIVATE Threads::Threads)
    add_executable(senti-bridge-test tests/failures.cpp bridge.cpp)
    target_compile_features(senti-bridge-test PRIVATE cxx_std_20)
    target_compile_options(senti-bridge-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_include_directories(senti-bridge-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(senti-bridge-test PRIVATE PkgConfig::JSONC Threads::Threads)
    add_test(NAME bridge-failures COMMAND senti-bridge-test "$<TARGET_FILE:senti-bad-host>" "${SENTI_HOST_EXECUTABLE}")
    set_tests_properties(bridge-failures PROPERTIES TIMEOUT 10)
    add_executable(senti-native-test tests/native.cpp)
    target_compile_features(senti-native-test PRIVATE cxx_std_20)
    target_compile_options(senti-native-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_include_directories(senti-native-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(senti-native-test PRIVATE senti Fcitx5::Core PkgConfig::JSONC ${CMAKE_DL_LIBS})
    add_test(NAME native-integration COMMAND senti-native-test "$<TARGET_FILE:senti>" "${SENTI_HOST_EXECUTABLE}")
    add_test(NAME native-backend-unavailable COMMAND senti-native-test "$<TARGET_FILE:senti>" "${CMAKE_CURRENT_BINARY_DIR}/does-not-exist" unavailable)
    set_tests_properties(native-backend-unavailable PROPERTIES ENVIRONMENT "SENTI_IME_EPHEMERAL=1;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/failure-config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/failure-data" TIMEOUT 10)
    set_tests_properties(native-integration PROPERTIES ENVIRONMENT "SENTI_IME_EPHEMERAL=1;XDG_CONFIG_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-config;XDG_DATA_HOME=${CMAKE_CURRENT_BINARY_DIR}/test-data" TIMEOUT 20)
endif()
